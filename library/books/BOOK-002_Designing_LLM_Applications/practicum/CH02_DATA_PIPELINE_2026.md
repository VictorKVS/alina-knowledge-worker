## Код первого уровня (только стандартная библиотека Python)

~~~python
import hashlib
import re

def norm(text):
    return re.sub(r'\s+', ' ', text).strip()

def sha(text):
    return hashlib.sha256(norm(text).encode('utf-8')).hexdigest()

assert sha('Данные  важны') == sha('Данные важны')
assert sha('Разные слова') != sha('Другая фраза')
print('Учебные проверки нормализации пройдены')
~~~
